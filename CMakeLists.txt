cmake_minimum_required(VERSION 3.21)
project(CornerStore VERSION 3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Multimedia)
find_package(ZXing REQUIRED CONFIG)

add_library(barcode_core src/Barcode.cpp src/ImageReader.cpp src/ScanGate.cpp)
target_include_directories(barcode_core PUBLIC src)
target_link_libraries(barcode_core PUBLIC Qt6::Widgets ZXing::ZXing)
add_library(pos_core src/Access.cpp src/Numbers.cpp src/RecordCodec.cpp src/Models.cpp
    src/Records.cpp src/ReadState.cpp src/TextStore.cpp src/Checkout.cpp src/Catalog.cpp
    src/Reports.cpp src/Receipt.cpp)
target_link_libraries(pos_core PUBLIC barcode_core Qt6::Network)
add_library(pos_ui src/CameraScanner.cpp src/MainWindow.cpp src/CashierPage.cpp src/ManagerPage.cpp)
target_link_libraries(pos_ui PUBLIC pos_core Qt6::Multimedia)
add_executable(CornerStore src/main.cpp)
target_link_libraries(CornerStore PRIVATE pos_ui)
set_target_properties(CornerStore PROPERTIES WIN32_EXECUTABLE TRUE)

include(CTest)
if(BUILD_TESTING)
    add_executable(barcode_tests tests/barcode_tests.cpp)
    target_link_libraries(barcode_tests PRIVATE barcode_core)
    add_test(NAME barcode_roundtrip COMMAND barcode_tests)
    foreach(suite IN ITEMS workflow storage ui)
        add_executable(${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(${suite}_tests PRIVATE pos_ui)
        add_test(NAME ${suite} COMMAND ${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/data")
    endforeach()
    set_tests_properties(barcode_roundtrip workflow storage ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 70)
endif()
