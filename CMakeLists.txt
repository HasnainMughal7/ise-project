cmake_minimum_required(VERSION 3.21)
project(CornerStore VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia)
find_package(ZXing REQUIRED CONFIG)

add_library(barcode_core src/Barcode.cpp src/ImageReader.cpp)
target_include_directories(barcode_core PUBLIC src)
target_link_libraries(barcode_core PUBLIC Qt6::Widgets ZXing::ZXing)
add_library(pos_core src/Numbers.cpp src/RecordCodec.cpp src/TextStore.cpp src/Checkout.cpp src/Receipt.cpp)
target_link_libraries(pos_core PUBLIC barcode_core)
add_library(pos_ui src/CameraScanner.cpp src/ScanDialog.cpp src/MainWindow.cpp)
target_link_libraries(pos_ui PUBLIC pos_core Qt6::Multimedia)
add_executable(CornerStore src/main.cpp)
target_link_libraries(CornerStore PRIVATE pos_ui)
set_target_properties(CornerStore PROPERTIES WIN32_EXECUTABLE TRUE)

include(CTest)
if(BUILD_TESTING)
    add_executable(barcode_tests tests/barcode_tests.cpp)
    target_link_libraries(barcode_tests PRIVATE barcode_core)
    add_test(NAME barcode_roundtrip COMMAND barcode_tests)
    add_executable(pos_tests tests/pos_tests.cpp)
    target_link_libraries(pos_tests PRIVATE pos_core)
    add_test(NAME pos_workflow COMMAND pos_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE pos_ui)
    add_test(NAME gui_workflow COMMAND ui_tests "${CMAKE_CURRENT_SOURCE_DIR}/data")
    set_tests_properties(barcode_roundtrip pos_workflow gui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
